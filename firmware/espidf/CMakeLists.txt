cmake_minimum_required(VERSION 3.16)
set(TRUCKBOX_ROLE "core" CACHE STRING "Firmware role: core or link")
if(NOT TRUCKBOX_ROLE STREQUAL "core" AND NOT TRUCKBOX_ROLE STREQUAL "link")
    message(FATAL_ERROR "TRUCKBOX_ROLE must be core or link")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.${TRUCKBOX_ROLE}.defaults")
set(PROJECT_VER "0.2.26" CACHE STRING "Application version included in OTA images")
set(IDF_TARGET "esp32c6")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(truckbox_${TRUCKBOX_ROLE})
if(TRUCKBOX_ROLE STREQUAL "core")
    if(NOT CONFIG_LWIP_IP_FORWARD OR NOT CONFIG_LWIP_IPV4_NAPT)
        message(FATAL_ERROR "Core requires effective IPv4 forwarding and NAPT; refusing a non-sharing image")
    endif()
    # The IDF default is 512 entries; constrain RAM for capture/HTTP on Core C6.
    idf_component_get_property(truckbox_lwip_lib lwip COMPONENT_LIB)
    target_compile_definitions(${truckbox_lwip_lib} PRIVATE IP_NAPT_MAX=64)
endif()
