cmake_minimum_required(VERSION 3.16)
option(TRUCKBOX_REVC_LILYGO_S3 "Build TruckBox Rev.C for LILYGO H707" OFF)
set(TRUCKBOX_ROLE "core" CACHE STRING "Firmware role: core or link")
if(NOT TRUCKBOX_ROLE STREQUAL "core" AND NOT TRUCKBOX_ROLE STREQUAL "link")
    message(FATAL_ERROR "TRUCKBOX_ROLE must be core or link")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
if(TRUCKBOX_REVC_LILYGO_S3)
    if(NOT TRUCKBOX_ROLE STREQUAL "core")
        message(FATAL_ERROR "Rev.C includes Core and Link and must use TRUCKBOX_ROLE=core")
    endif()
    set(IDF_TARGET "esp32s3")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.revc.defaults")
    set(PROJECT_VER "0.1.0-revc" CACHE STRING "Application version included in OTA images")
    set(truckbox_project truckbox_revc_lilygo_s3)
else()
    set(IDF_TARGET "esp32c6")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.${TRUCKBOX_ROLE}.defaults")
    set(PROJECT_VER "0.2.26" CACHE STRING "Application version included in OTA images")
    set(truckbox_project truckbox_${TRUCKBOX_ROLE})
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(TRUCKBOX_REVC_LILYGO_S3)
    idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.revc.lock")
endif()
project(${truckbox_project})
if(TRUCKBOX_REVC_LILYGO_S3)
    if(NOT CONFIG_TB_BOARD_REVC OR NOT CONFIG_TB_ROLE_CORE OR NOT CONFIG_TB_J1708_ONLY
       OR NOT CONFIG_ESPTOOLPY_FLASHSIZE_16MB OR NOT CONFIG_SPIRAM_MODE_OCT
       OR NOT CONFIG_PARTITION_TABLE_CUSTOM_FILENAME STREQUAL "partitions_revc.csv")
        message(FATAL_ERROR "Rev.C requires the H707 16 MB/OPI PSRAM profile and its partition table. Use a fresh build directory.")
    endif()
elseif(CONFIG_TB_BOARD_REVC)
    message(FATAL_ERROR "Select TRUCKBOX_REVC_LILYGO_S3 and use a fresh Rev.C build directory")
endif()
if(TRUCKBOX_ROLE STREQUAL "core")
    if(NOT CONFIG_LWIP_IP_FORWARD OR NOT CONFIG_LWIP_IPV4_NAPT)
        message(FATAL_ERROR "Core requires effective IPv4 forwarding and NAPT; refusing a non-sharing image")
    endif()
    # The IDF default is 512 entries; constrain RAM for capture/HTTP on Core C6.
    idf_component_get_property(truckbox_lwip_lib lwip COMPONENT_LIB)
    target_compile_definitions(${truckbox_lwip_lib} PRIVATE IP_NAPT_MAX=64)
endif()
